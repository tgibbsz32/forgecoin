cmake_minimum_required(VERSION 3.22.1)
project(forgecoin_android_runtime LANGUAGES CXX)
add_library(forge_runtime_probe SHARED forge_runtime_probe.cpp)
target_compile_features(forge_runtime_probe PRIVATE cxx_std_17)
target_compile_options(forge_runtime_probe PRIVATE -Wall -Wextra -Werror)

# Use the fork's vendored RandomX source; never substitute a generic algorithm.
set(RANDOMX_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../external/randomx")
if(NOT EXISTS "${RANDOMX_SOURCE_DIR}/src/randomx.h")
  message(FATAL_ERROR "ForgeCoin vendored RandomX sources missing; fetch external/randomx")
endif()
set(ARCH_ID "aarch64")
set(ARM_ID "aarch64")
add_subdirectory("${RANDOMX_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/randomx-build" EXCLUDE_FROM_ALL)
add_library(forge_randomx SHARED randomx_bridge.cpp)
target_include_directories(forge_randomx PRIVATE "${RANDOMX_SOURCE_DIR}/src")
target_link_libraries(forge_randomx PRIVATE randomx)
target_compile_features(forge_randomx PRIVATE cxx_std_17)
