cmake_minimum_required(VERSION 3.18)
project(forgecoin_android_wallet LANGUAGES C CXX)
get_filename_component(FORGE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_GUI_DEPS ON CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_DOCUMENTATION OFF CACHE BOOL "" FORCE)
set(USE_DEVICE_TREZOR OFF CACHE BOOL "" FORCE)
set(USE_READLINE OFF CACHE BOOL "" FORCE)
set(STACK_TRACE OFF CACHE BOOL "" FORCE)
set(PER_BLOCK_CHECKPOINT OFF CACHE BOOL "" FORCE)
add_subdirectory("${FORGE_SOURCE}" core EXCLUDE_FROM_ALL)
add_library(forge_wallet SHARED ../app/src/main/cpp/wallet_bridge.cpp)
target_compile_features(forge_wallet PRIVATE cxx_std_14)
target_include_directories(forge_wallet PRIVATE "${FORGE_SOURCE}/src")
target_link_libraries(forge_wallet PRIVATE wallet_api)
